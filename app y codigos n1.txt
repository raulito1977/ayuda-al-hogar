-- =============================================
-- AYUDAALHOGAR.COM
-- ESQUEMA DE PRODUCCION V1
-- =============================================
 
-- =============================================
-- USUARIOS
-- =============================================
 
create table usuarios (
 
id uuid primary key default gen_random_uuid(),
 
email varchar(255) unique not null,
 
password_hash text not null,
 
rol varchar(30) not null,
 
ip_registro varchar(100),
 
activo boolean default true,
 
email_verificado boolean default false,
 
fecha_registro timestamp default now(),
 
constraint chk_rol
check (
rol in (
'familia',
'voluntario',
'profesional',
'admin'
)
)
 
);
 
-- =============================================
-- PERFILES
-- =============================================
 
create table perfiles (
 
id uuid primary key default gen_random_uuid(),
 
usuario_id uuid not null
references usuarios(id)
on delete cascade,
 
nombre varchar(150) not null,
 
telefono varchar(50),
 
direccion text,
 
ciudad varchar(100),
 
provincia varchar(100),
 
codigo_postal varchar(20),
stripe_account_id varchar (255),
tipo_perfil varchar(50),
verificado boolean default false,
 
foto_url text,
 
biografia text,
 
fecha_creacion timestamp default now()
 
);
 
-- =============================================
-- UBICACIONES GPS
-- =============================================
 
create table ubicaciones (
 
id uuid primary key default gen_random_uuid(),
 
usuario_id uuid not null
references usuarios(id)
on delete cascade,
 
latitud decimal(10,8),
 
longitud decimal(11,8),
 
actualizado timestamp default now()
 
);
 
-- =============================================
-- SERVICIOS DISPONIBLES
-- =============================================
 
create table servicios (
 
id uuid primary key default gen_random_uuid(),
 
nombre varchar(150) not null,
 
descripcion text,
 
categoria varchar(100),
 
activo boolean default true,
 
fecha_creacion timestamp default now()
 
);
 
-- =============================================
-- ALERTAS
-- =============================================
 
create table alertas (
 
id uuid primary key default gen_random_uuid(),
 
usuario_id uuid not null
references usuarios(id),
 
servicio_id uuid
references servicios(id),
 
titulo varchar(255) not null,
 
descripcion text,
 
fecha_alerta timestamp default now(),
 
estado varchar(30) default 'abierta',
 
profesional_id uuid
references usuarios(id),
 
constraint chk_estado_alerta
check (
estado in (
'abierta',
'aceptada',
'en_proceso',
'finalizada',
'cancelada'
)
)
 
);
 
-- =============================================
-- PAGOS MARKETPLACE
-- =============================================
 
create table pagos (
 
id uuid primary key default gen_random_uuid(),
 
alerta_id uuid
references alertas(id),
 
cliente_id uuid
references usuarios(id),
 
profesional_id uuid
references usuarios(id),
 
stripe_payment_intent_id text,
 
importe_total decimal(10,2),
 
porcentaje_comision decimal(5,2) default 15.00,
 
importe_comision decimal(10,2),
 
importe_profesional decimal(10,2),
 
estado varchar(30) default 'pendiente',
 
fecha_pago timestamp,
 
constraint chk_estado_pago
check (
estado in (
'pendiente',
'pagado',
'cancelado',
'reembolsado'
)
)
 
);
 
-- =============================================
-- STRIPE CONNECT
-- =============================================
 
create table stripe_connect (
 
id uuid primary key default gen_random_uuid(),
 
usuario_id uuid unique
references usuarios(id),
 
stripe_account_id text,
 
onboarding_completado boolean default false,
 
fecha_creacion timestamp default now()
 
);
 
-- =============================================
-- VALORACIONES
-- =============================================
 
create table valoraciones (
 
id uuid primary key default gen_random_uuid(),
 
emisor_id uuid not null
references usuarios(id),
 
receptor_id uuid not null
references usuarios(id),
 
estrellas integer not null,
 
comentario text,
 
fecha_valoracion timestamp default now(),
 
constraint chk_estrellas
check (
estrellas >= 1
and estrellas <= 5
)
 
);
 
-- =============================================
-- MENSAJES INTERNOS
-- =============================================
 
create table mensajes (
 
id uuid primary key default gen_random_uuid(),
 
remitente_id uuid
references usuarios(id),
 
destinatario_id uuid
references usuarios(id),
 
mensaje text not null,
 
fecha_envio timestamp default now()
 
);
 
-- =============================================
-- INDICES
-- =============================================
 
create index idx_usuario_rol
on usuarios(rol);
 
create index idx_alerta_estado
on alertas(estado);
 
create index idx_pago_estado
on pagos(estado);
 
create index idx_ubicacion_usuario
on ubicaciones(usuario_id);
 
create index idx_valoracion_receptor
on valoraciones(receptor_id);
